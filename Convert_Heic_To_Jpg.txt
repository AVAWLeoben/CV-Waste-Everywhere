from pathlib import Path
from PIL import Image, ImageOps
from pillow_heif import register_heif_opener


register_heif_opener()

input_dir = Path(r"C:\Users\GKoinig\Desktop\CV-Waste-Everywhere\Data")
output_dir = Path(r"C:\Users\GKoinig\Desktop\CV-Waste-Everywhere\Data")
output_dir.mkdir(parents=True, exist_ok=True)

for source in input_dir.iterdir():
    if source.suffix.lower() not in {".heic", ".heif"}:
        continue

    target = output_dir / f"{source.stem}.jpg"

    try:
        with Image.open(source) as image:
            image = ImageOps.exif_transpose(image).convert("RGB")
            # Remove metadata after applying orientation.
            image.info.clear()
            # Image.save replaces an existing JPEG at this path.
            image.save(target, "JPEG", quality=95, subsampling=0)

        source.unlink()  # Delete the HEIC only after successful conversion.
        print(f"Converted: {source.name} -> {target}")

    except Exception as error:
        print(f"Failed: {source.name}: {error}")
